  METHOD cancel_challan.
    DATA: WA_CANCEL_EWAY_REQ TYPE TY_CANCEL_REQUEST,
          LT_ERROR_RESPONSE  TYPE STANDARD TABLE OF TY_PROPERTY_ERROR,
          LV_ERROR           TYPE string,
          LT_UPDATE          TYPE TABLE FOR UPDATE zmm_c_challan.
    IF keys IS NOT INITIAL.
      SELECT document_number,document_date,company_code,fiscal_year,location_gstin,billfromgstin,documenttype
        FROM zsd_t_einv_log
        FOR ALL ENTRIES IN @keys
        WHERE document_number = @keys-DocumentNumber
        AND ewaybill_number IS NOT INITIAL
        INTO TABLE @DATA(LT_EINV_LOG).
      IF LT_EINV_LOG IS NOT INITIAL.
        WA_CANCEL_EWAY_REQ-reason = 'Duplicate'.
        WA_CANCEL_EWAY_REQ-remarks = 'Duplicate'.
        WA_CANCEL_EWAY_REQ-criterias = VALUE #( FOR WA_EINV_LOG IN LT_EINV_LOG ( location_gstin = WA_EINV_LOG-location_gstin
                                                                                 document_number = |{ WA_EINV_LOG-document_number ALPHA = OUT }|
                                                                                 document_date = |{ WA_EINV_LOG-Document_Date+6(2) }-{ WA_EINV_LOG-Document_Date+4(2) }-{ WA_EINV_LOG-Document_Date+0(4) }|
                                                                                 supply_type = 'S'
                                                                                 document_type = WA_EINV_LOG-documenttype
                                                                                 bill_from_gstin = WA_EINV_LOG-billfromgstin ) ).
        DATA(LV_CANCEL_REQ) = /ui2/cl_json=>serialize( data = WA_CANCEL_EWAY_REQ
                                                       pretty_name = /ui2/cl_json=>pretty_mode-camel_case ).
        DATA(LO_INVOICE) = NEW zcl_invoice(  ).
        LO_INVOICE->call_cpi( EXPORTING iv_req_payload = LV_CANCEL_REQ
                                        iv_action      = 'CancelEwaybill'
                                        IMPORTING ev_response    = DATA(LV_REFERENCE) ).
        FIND REGEX '"referenceId"\s*:\s*"([^"]+)"' IN LV_REFERENCE SUBMATCHES DATA(LV_REFERENCEID).
        IF LV_REFERENCEID IS NOT INITIAL.
          LO_INVOICE->call_cpi( EXPORTING  iv_action       = 'Status'
                                           iv_reference_id = LV_REFERENCEID
                                IMPORTING  ev_response     = DATA(LV_RESPONSE) ).
          FIND REGEX '"status"\s*:\s*"([^"]*)"' IN LV_RESPONSE SUBMATCHES DATA(LV_STATUS).
          IF LV_STATUS EQ 'P'.
            LT_UPDATE = VALUE #( FOR WA_KEYS IN keys ( DocumentNumber = WA_KEYS-DocumentNumber
                                   DocumentDate = WA_KEYS-DocumentDate
                                   Company_Code = WA_KEYS-Company_Code
                                   EwayCancel = 'X'
                                   EwayFail = ''
                                   EwaybillError = ''
                                   %control-DocumentNumber = if_abap_behv=>mk-on
                                   %control-DocumentDate = if_abap_behv=>mk-on
                                   %control-Company_Code = if_abap_behv=>mk-on
                                   %control-EwayFail = if_abap_behv=>mk-on
                                   %control-EwaybillError = if_abap_behv=>mk-on
                                   %control-EwayCancel = if_abap_behv=>mk-on ) ).
            reported-zmm_c_challan = VALUE #( (
                   %msg = new_message( id = 'EWB'
                                       number = '1'
                                       severity = if_abap_behv_message=>severity-information
                                       v1 = 'EWB CANCEL SUCCESSFULLY'
                                      )
) ).
          ENDIF.
        ELSE.
          /ui2/cl_json=>deserialize( EXPORTING json = LV_REFERENCE
                                     CHANGING  data = LT_ERROR_RESPONSE ).
          TRY.
              DATA(WA_ERROR_RESPONSE) = LT_ERROR_RESPONSE[ 1 ].
              IF WA_ERROR_RESPONSE-errors IS NOT INITIAL.
                LOOP AT WA_ERROR_RESPONSE-errors INTO DATA(WA_ERRORS).
                  LV_ERROR = |{ LV_ERROR } | && '|' &&  | { WA_ERRORS-message }|.
                  IF sy-tabix EQ '1'.
                    REPLACE ALL OCCURRENCES OF '|' IN LV_ERROR WITH ''.
                  ENDIF.
                ENDLOOP.
                LT_UPDATE = VALUE #( FOR WA_KEYS IN keys ( DocumentNumber = WA_KEYS-DocumentNumber
                                                           DocumentDate = WA_KEYS-DocumentDate
                                                           Company_Code = WA_KEYS-Company_Code
                                                           EwaybillError = LV_ERROR
                                                           %control-DocumentNumber = if_abap_behv=>mk-on
                                                           %control-DocumentDate = if_abap_behv=>mk-on
                                                           %control-Company_Code = if_abap_behv=>mk-on
                                                           %control-EwaybillError = if_abap_behv=>mk-on ) ).
                reported-zmm_c_challan = VALUE #( ( %msg = new_message_with_text( text = LV_ERROR
                                     severity = if_abap_behv_message=>severity-error ) )  ).
              ENDIF.
            CATCH cx_sy_itab_line_not_found.
          ENDTRY.
        ENDIF.
      ELSE.
        reported-zmm_c_challan = VALUE #( ( %msg = new_message( id = 'ZSD_MSG_CYG'
                                                                  number = '003'
                                                                  severity = if_abap_behv_message=>severity-error
                                                                  v1 = keys[ 1 ]-DocumentNumber ) )  ) .
      ENDIF.
      IF LT_UPDATE IS NOT INITIAL.
        MODIFY ENTITIES OF zmm_c_challan IN LOCAL MODE
         ENTITY zmm_c_challan
         UPDATE FROM LT_UPDATE
         FAILED DATA(LT_FAILED)
         REPORTED DATA(LT_REPORTED)
         MAPPED DATA(LT_MAPPED).
      ENDIF.
    ENDIF.
  ENDMETHOD.

            UPDATE FROM LT_UPDATE_LOG
            FAILED DATA(LT_FAILED)
            REPORTED DATA(LT_REPORTED)
            MAPPED DATA(LT_MAPPED).
        ENDIF.
      ELSE.
        reported-zmm_c_challan = VALUE #( ( %msg = new_message( id = 'ZSD_MSG_CYG'
                                                                number = '002'
                                                                severity = if_abap_behv_message=>severity-error
                                                                v1 = keys[ 1 ]-DocumentNumber ) )  ) .
      ENDIF.
    ENDIF.
  ENDMETHOD.

  METHOD download_pdf.
    DATA: LT_FINAL_TEMP TYPE zsd_tt_ewb_status,
          LT_FORM_A     TYPE zsd_tt_ewb_part_a,
          LT_FORM_QR    TYPE zsd_tt_ewb_part_a,
          LT_FORM_EWB   TYPE zsd_tt_ewb_part_a,
          LT_FORM_IRN   TYPE zsd_tt_ewb_part_a,
          LT_FORM_B     TYPE zsd_tt_ewb_part_b.
    DATA: LV_FM_NAME TYPE rs38l_fnam,
          WA_CONTROL TYPE ssfctrlop,
          WA_OUTPUT  TYPE ssfcrescl.
    LV_FM_NAME = '/1BCDWB/SF00000097'.
    WA_CONTROL-no_dialog = abap_true.
    WA_CONTROL-preview   = abap_false.
    WA_CONTROL-getotf    = abap_true.
    CALL FUNCTION LV_FM_NAME
      EXPORTING
        control_parameters = WA_CONTROL
        user_settings      = 'X'
      IMPORTING
        job_output_info    = WA_OUTPUT
      TABLES
        gt_data            = LT_FINAL_TEMP
        gt_form_qr         = LT_FORM_QR
        gt_form_ewb        = LT_FORM_EWB
        gt_form_irn        = LT_FORM_IRN
        gt_form_a          = LT_FORM_A
        gt_form_b          = LT_FORM_B
      EXCEPTIONS
        formatting_error   = 1
        internal_error     = 2
        send_error         = 3
        user_canceled      = 4
        OTHERS             = 5.
    DATA: LT_OTF        TYPE STANDARD TABLE OF itcoo,
          LT_LINES      TYPE STANDARD TABLE OF tline,
          LV_PDF        TYPE xstring,
          LV_PDF_BASE64 TYPE string.
    LT_OTF = WA_OUTPUT-otfdata.
    CALL FUNCTION 'CONVERT_OTF'
      EXPORTING
        format                = 'PDF'
      IMPORTING
        bin_file              = LV_PDF
      TABLES
        otf                   = LT_OTF
        lines                 = LT_LINES
      EXCEPTIONS
        err_max_linewidth     = 1
        err_format            = 2
        err_conv_not_possible = 3
        err_bad_otf           = 4
        OTHERS                = 5.
    IF sy-subrc <> 0.
      RETURN.
    ENDIF.
    LV_PDF_BASE64 = cl_http_utility=>encode_x_base64( LV_PDF ).
    result = VALUE #( (
           %tky = keys[ 1 ]-%tky
        %param-FileName = |{ keys[ 1 ]-DocumentNumber }.pdf|
       %param-mimeType = 'application/pdf'
          %param-pdf_data = LV_PDF_BASE64
       ) ).
  ENDMETHOD.
ENDCLASS.
